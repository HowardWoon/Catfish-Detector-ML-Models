def train_models(x_train: np.ndarray, y_train: np.ndarray) -> Dict[str, Any]:
    positive_weight = float((y_train == 0).sum() / max((y_train == 1).sum(), 1))
    
    base_models = {
        "Logistic Regression": LogisticRegression(max_iter=3000, solver="saga", class_weight="balanced"),
        "Decision Tree": DecisionTreeClassifier(class_weight="balanced", max_features="sqrt"),
        "Random Forest": RandomForestClassifier(class_weight="balanced_subsample", max_features="sqrt"),
        "Extra Trees": ExtraTreesClassifier(class_weight="balanced_subsample", max_features="sqrt"),
        "XGBoost": XGBClassifier(scale_pos_weight=positive_weight, eval_metric="auc", tree_method="hist"),
        "MLP Neural Network": MLPClassifier(early_stopping=True, max_iter=500)
    }

    param_grids = {
        "Logistic Regression": {"C": [0.01, 0.1, 0.5, 1, 5], "penalty": ["l2"]},
        "Decision Tree": {"max_depth": [5, 10, 15, None], "min_samples_split": [5, 10, 20]},
        "Random Forest": {"n_estimators": [100, 200, 300], "max_depth": [10, 15, 20]},
        "Extra Trees": {"n_estimators": [100, 200, 300], "max_depth": [10, 15, 20]},
        "XGBoost": {"n_estimators": [100, 200, 300], "learning_rate": [0.03, 0.05, 0.1]},
        "MLP Neural Network": {"hidden_layer_sizes": [(128, 64), (256, 128, 64)], "alpha": [0.0001, 0.001]}
    }

    cv = StratifiedKFold(n_splits=3, shuffle=True, random_state=42)
    tuned_models = {}

    for name, model in base_models.items():
        rs = RandomizedSearchCV(model, param_grids[name], n_iter=5, cv=cv, scoring="f1_macro", n_jobs=-1)
        rs.fit(x_train, y_train)
        tuned_models[name] = rs.best_estimator_

    return tuned_models
